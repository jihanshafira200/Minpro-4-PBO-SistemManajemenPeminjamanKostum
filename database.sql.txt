CREATE DATABASE IF NOT EXISTS db_peminjaman_kostum;

USE db_peminjaman_kostum;

CREATE TABLE kostum (
    id_kostum VARCHAR(10) PRIMARY KEY,
    nama VARCHAR(100) NOT NULL,
    kategori VARCHAR(50) NOT NULL,
    ukuran INT NOT NULL,
    harga_sewa DECIMAL(12,2) NOT NULL
);

CREATE TABLE pelanggan (
    id_pelanggan VARCHAR(10) PRIMARY KEY,
    nama VARCHAR(100) NOT NULL,
    nomor_telepon VARCHAR(20) NOT NULL,
    alamat VARCHAR(255) NOT NULL
);

CREATE TABLE peminjaman (
    id_peminjaman VARCHAR(10) PRIMARY KEY,
    id_pelanggan VARCHAR(10) NOT NULL,
    id_kostum VARCHAR(10) NOT NULL,
    tanggal_pinjam DATE NOT NULL,
    tanggal_pengembalian DATE NOT NULL,
    status_peminjaman VARCHAR(20) NOT NULL,

    CONSTRAINT fk_peminjaman_pelanggan
        FOREIGN KEY (id_pelanggan)
        REFERENCES pelanggan(id_pelanggan),

    CONSTRAINT fk_peminjaman_kostum
        FOREIGN KEY (id_kostum)
        REFERENCES kostum(id_kostum)
);
